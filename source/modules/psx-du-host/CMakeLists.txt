cmake_minimum_required(VERSION 3.30)
project(psx_du_host VERSION 1.0.0 LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../shared/windows-version/VersionInfo.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../shared/windows-icons/WindowsIcons.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DU_HOST_GRAPHICS144 TRUE)
option(PSX_DU_HOST_INHERITED_GRAPHICS144 "Explicit qualified ASS1.4.5 inherited consumer" OFF)
set(PSX_DU_HOST_GRAPHICS144_SDK_ROOT "$ENV{ASS_GRAPHICS_SDK_ROOT}" CACHE PATH "Qualified complete Debug/Release ASS1.4.5 SDK root")
if(NOT PSX_DU_HOST_GRAPHICS144_SDK_ROOT)
 message(FATAL_ERROR "Set PSX_DU_HOST_GRAPHICS144_SDK_ROOT to the qualified full SDK root")
endif()
find_package(AssGraphics 1.4.5 EXACT CONFIG REQUIRED)
add_subdirectory(../../shared/cdu cdu)
add_subdirectory(../../shared/display-transport display-transport)
add_subdirectory(../../shared/diagnostics diagnostics)
add_library(du_host_core STATIC src/config.cpp src/runtime.cpp)
target_include_directories(du_host_core PUBLIC include)
target_link_libraries(du_host_core PUBLIC psx_cdu_transport psx_display_transport psx_diagnostics ass::graphics ole32)
add_library(du_host_ui STATIC src/ui.cpp)
target_include_directories(du_host_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../shared/windows-icons")
target_link_libraries(du_host_ui PUBLIC du_host_core)
add_executable(psx-du-host WIN32 src/main.cpp app.manifest)
target_link_libraries(psx-du-host PRIVATE du_host_ui shell32)
psx_target_icon(psx-du-host DU)
foreach(target du_host_core du_host_ui psx-du-host)
 target_compile_options(${target} PRIVATE /W4 /Zi /utf-8 /EHsc)
 target_compile_definitions(${target} PRIVATE PSX_DU_HOST_VERSION="${PROJECT_VERSION}" PSX_DU_HOST_VERSION_WIDE=L\"${PROJECT_VERSION}\" UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endforeach()
target_link_options(psx-du-host PRIVATE /DEBUG /INCREMENTAL:NO)
add_custom_command(TARGET psx-du-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:ass::graphics>" "$<TARGET_FILE_DIR:psx-du-host>" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../../shared/norm/vendor/norm/runtime" "$<TARGET_FILE_DIR:psx-du-host>")
enable_testing()
add_executable(du_host_test tests/host_test.cpp app.manifest)
target_link_libraries(du_host_test PRIVATE du_host_ui psapi)
if(DU_HOST_GRAPHICS144)
 target_compile_definitions(du_host_test PRIVATE DU_HOST_GRAPHICS144=1)
 target_link_libraries(du_host_test PRIVATE comctl32)
 add_test(NAME du_host_graphics144 COMMAND du_host_test graphics144 "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/graphics144")
 set_tests_properties(du_host_graphics144 PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)

endif()
psx_target_icon(du_host_test DU)
target_compile_options(du_host_test PRIVATE /Zi /W4 /utf-8 /EHsc)
target_compile_definitions(du_host_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_options(du_host_test PRIVATE /DEBUG /INCREMENTAL:NO)
add_custom_command(TARGET du_host_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:ass::graphics>" "$<TARGET_FILE_DIR:du_host_test>" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../../shared/norm/vendor/norm/runtime" "$<TARGET_FILE_DIR:du_host_test>")
foreach(operation source-observability config runtime ui diagnostics continuity local blackout)
 add_test(NAME du_host_${operation} COMMAND du_host_test ${operation} "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/${operation}")
 set_tests_properties(du_host_${operation} PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)
endforeach()
find_program(DU_PACKAGE_PWSH pwsh REQUIRED)
add_test(NAME du_host_package_contract COMMAND "${DU_PACKAGE_PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/package_contract.ps1")
