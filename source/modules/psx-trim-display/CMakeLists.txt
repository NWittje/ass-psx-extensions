cmake_minimum_required(VERSION 3.22)
if(POLICY CMP0141)
  cmake_policy(SET CMP0141 NEW)
endif()
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(psx_trim_host_qualification LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT EXISTS "${TRIM_ASSET_CPP}")
  message(FATAL_ERROR "Generate native artwork before configuring host fixtures")
endif()
add_subdirectory(../../shared/hardware-link hardware-link)
add_library(trim_core STATIC core/protocol.cpp core/state.cpp core/application.cpp core/scene.cpp "${TRIM_ASSET_CPP}")
target_compile_definitions(trim_core PRIVATE TRIM_BUILD_ROLE=0)
target_include_directories(trim_core PUBLIC include)
target_link_libraries(trim_core PUBLIC psx_hardware_link)
add_executable(trim_qualification tests/qualification.cpp)
target_link_libraries(trim_qualification PRIVATE trim_core)
add_executable(trim_render_demo tools/render_demo.cpp)
target_link_libraries(trim_render_demo PRIVATE trim_core)
foreach(target psx_hardware_link trim_core trim_qualification trim_render_demo)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
add_test(NAME trim_qualification COMMAND trim_qualification "${TRIM_REFERENCE_DIR}")
