file(STRINGS "${lzma_root}/SHA256SUMS" lzma_checks)
foreach(row IN LISTS lzma_checks)
 string(SUBSTRING "${row}" 0 64 expected)
 string(SUBSTRING "${row}" 66 -1 name)
 file(SHA256 "${lzma_root}/${name}" actual)
 if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "LZMA decoder input changed: ${name}")
 endif()
endforeach()

function(psx_plugin_compile_settings target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_include_directories(${target} PRIVATE include "${XPLANE_SDK_ROOT}/CHeaders/XPLM")
  target_compile_definitions(${target} PRIVATE IBM=1 APL=0 LIN=0 XPLM200 XPLM210 XPLM300 XPLM301 XPLM303 XPLM400 XPLM410 XPLM420 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHs /utf-8)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endfunction()
