cmake_minimum_required(VERSION 3.30)
project(psx_du_client VERSION 0.4.12 LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../shared/windows-version/VersionInfo.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/../../shared/windows-icons/WindowsIcons.cmake")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(ASS_GRAPHICS_PACKAGE)
 find_package(AssGraphics 1.4.2 EXACT CONFIG REQUIRED PATHS "${ASS_GRAPHICS_PACKAGE}/lib/cmake/AssGraphics" NO_DEFAULT_PATH)
else()
 if(NOT ASS_GRAPHICS_SOURCE)
  set(ASS_GRAPHICS_SOURCE "P:/development/ass_graphics_lib")
 endif()
 set(ASS_GRAPHICS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(ASS_GRAPHICS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 add_subdirectory("${ASS_GRAPHICS_SOURCE}" ass-graphics)
endif()
add_subdirectory(../../shared/display-replay display-replay)
add_subdirectory(../../shared/display-transport display-transport)
add_library(du_client_core STATIC src/runtime.cpp src/config.cpp src/diagnostics.cpp src/trim.cpp src/failure_evidence.cpp)
target_include_directories(du_client_core PUBLIC include)
target_link_libraries(du_client_core PUBLIC psx_display_replay psx_display_transport advapi32)
add_library(du_client_ui STATIC src/ui.cpp)
target_include_directories(du_client_ui PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../../shared/windows-icons")
target_link_libraries(du_client_ui PUBLIC du_client_core shell32 comctl32)
add_executable(psx-du-client WIN32 src/main.cpp app.manifest)
target_link_libraries(psx-du-client PRIVATE du_client_ui)
psx_target_icon(psx-du-client DU)
add_executable(du_client_test tests/test.cpp tests/runtime_test.cpp app.manifest)
add_executable(du_trim_test tests/trim.cpp app.manifest)
add_executable(du_input_exit_test tests/input_exit.cpp app.manifest)
target_link_libraries(du_input_exit_test PRIVATE du_client_ui)
psx_target_icon(du_input_exit_test DU)
target_compile_definitions(du_input_exit_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(du_input_exit_test PRIVATE /W4 /Zi /utf-8 /EHsc)
target_link_libraries(du_trim_test PRIVATE du_client_ui)
psx_target_icon(du_trim_test DU)
target_compile_definitions(du_trim_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(du_trim_test PRIVATE /W4 /Zi /utf-8 /EHsc)
target_link_options(du_trim_test PRIVATE /DEBUG /INCREMENTAL:NO)
target_link_libraries(du_client_test PRIVATE du_client_ui windowscodecs ole32 psapi)
psx_target_icon(du_client_test DU)
foreach(t du_client_core du_client_ui psx-du-client du_client_test psx_display_replay)
 target_compile_definitions(${t} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_compile_options(${t} PRIVATE /W4 /Zi /utf-8 /EHsc)
endforeach()
foreach(t psx-du-client du_client_test du_trim_test du_input_exit_test)
 target_link_options(${t} PRIVATE /DEBUG /INCREMENTAL:NO)
 add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:ass::graphics>" "$<TARGET_FILE_DIR:${t}>")
endforeach()
# The stall fixture exercises independent legacy CDU servicing. Production DU
# acquisition and its lean runtime package have no NORM dependency.
add_custom_command(TARGET du_client_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../../shared/norm/vendor/norm/runtime" "$<TARGET_FILE_DIR:du_client_test>")

enable_testing()
add_test(NAME du_reconnect-runtime COMMAND du_client_test reconnect-runtime "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/reconnect-runtime")
set_tests_properties(du_reconnect-runtime PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
add_test(NAME du_silent-runtime COMMAND du_client_test silent-runtime "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/silent-runtime")
set_tests_properties(du_silent-runtime PROPERTIES TIMEOUT 40 RUN_SERIAL TRUE)
foreach(mode input close)
 add_test(NAME du_queued_${mode} COMMAND du_input_exit_test ${mode} "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/queued-${mode}")
 set_tests_properties(du_queued_${mode} PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
endforeach()
add_test(NAME du_trim_ui COMMAND du_trim_test "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/trim-ui")
set_tests_properties(du_trim_ui PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
foreach(mode observability model config ui rates ui-runtime runtime network-runtime tcp-stall owned-source owned-budget)
 add_test(NAME du_${mode} COMMAND du_client_test ${mode} "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/${mode}")
 set_tests_properties(du_${mode} PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
endforeach()
if(PSX_DU_NORM_QUALIFICATION)
 add_test(NAME du_norm_adapter_stall COMMAND du_client_test network-stall "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/norm-stall")
 set_tests_properties(du_norm_adapter_stall PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
endif()

psx_stamp_directory("${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_VERSION}")
