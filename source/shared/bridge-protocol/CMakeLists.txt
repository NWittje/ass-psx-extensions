cmake_minimum_required(VERSION 3.24)
if(NOT TARGET psx_aircraft_state)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../aircraft-state" "${CMAKE_CURRENT_BINARY_DIR}/aircraft-state")
endif()
add_library(psx_bridge_protocol STATIC src/protocol.cpp)
target_include_directories(psx_bridge_protocol PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
if(NOT TARGET psx_motion)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../motion" "${CMAKE_CURRENT_BINARY_DIR}/motion")
endif()
target_include_directories(psx_bridge_protocol PUBLIC "${CMAKE_CURRENT_LIST_DIR}/../diagnostics/include")
target_link_libraries(psx_bridge_protocol PUBLIC psx_aircraft_state psx_motion)
target_compile_features(psx_bridge_protocol PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(psx_bridge_protocol PRIVATE /W4 /WX /permissive- /utf-8)
endif()

if(NOT TARGET psx_audio)
 add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../audio" "${CMAKE_CURRENT_BINARY_DIR}/audio")
endif()
target_link_libraries(psx_bridge_protocol PUBLIC psx_audio)

if(NOT TARGET psx_scenery_coverage)
 add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../scenery-coverage" "${CMAKE_CURRENT_BINARY_DIR}/scenery-coverage")
endif()
target_link_libraries(psx_bridge_protocol PUBLIC psx_scenery_coverage)
