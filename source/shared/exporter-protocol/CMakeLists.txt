cmake_minimum_required(VERSION 3.24)
if(NOT TARGET psx_motion)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../motion" "${CMAKE_CURRENT_BINARY_DIR}/motion")
endif()
add_library(psx_exporter_protocol STATIC src/protocol.cpp)
target_include_directories(psx_exporter_protocol PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_link_libraries(psx_exporter_protocol PUBLIC psx_motion)
target_compile_features(psx_exporter_protocol PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(psx_exporter_protocol PRIVATE /W4 /WX /permissive- /utf-8)
endif()

if(NOT TARGET psx_audio)
 add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../audio" "${CMAKE_CURRENT_BINARY_DIR}/audio")
endif()
target_link_libraries(psx_exporter_protocol PUBLIC psx_audio)
