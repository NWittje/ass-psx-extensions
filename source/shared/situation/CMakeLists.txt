cmake_minimum_required(VERSION 3.24)
project(psx_situation VERSION 0.2.0 LANGUAGES CXX)
if(NOT TARGET psx_non_normal)
  add_subdirectory(../non-normal "${CMAKE_CURRENT_BINARY_DIR}/non-normal")
endif()
add_library(psx_situation STATIC src/snapshot.cpp src/protocol.cpp src/client.cpp src/fmc.cpp src/fmc_profile.cpp src/fmc_protocol.cpp src/fmc_service.cpp src/fmc_native.cpp src/fmc_native_profile.cpp src/fmc_native_service.cpp)
target_compile_features(psx_situation PUBLIC cxx_std_20)
target_include_directories(psx_situation PUBLIC include)
target_compile_definitions(psx_situation PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(psx_situation PUBLIC psx_non_normal PRIVATE bcrypt)
option(PSX_SITUATION_TESTS "Build situation infrastructure qualification" ON)
if(PSX_SITUATION_TESTS)
  enable_testing()
  add_executable(psx_situation_tests tests/snapshot_tests.cpp)
  target_link_libraries(psx_situation_tests PRIVATE psx_situation)
  add_test(NAME psx_situation_snapshot COMMAND psx_situation_tests)
  add_executable(psx_fmc_snapshot_tests tests/fmc_tests.cpp)
  target_link_libraries(psx_fmc_snapshot_tests PRIVATE psx_situation)
  add_test(NAME psx_fmc_snapshot COMMAND psx_fmc_snapshot_tests)
  add_executable(psx_fmc_native_records_tests tests/fmc_native_tests.cpp)
  target_link_libraries(psx_fmc_native_records_tests PRIVATE psx_situation)
  add_test(NAME psx_fmc_native_records COMMAND psx_fmc_native_records_tests)
  add_executable(psx_fmc_native_service_tests tests/fmc_native_service_tests.cpp)
  target_link_libraries(psx_fmc_native_service_tests PRIVATE psx_situation)
  add_test(NAME psx_fmc_native_service COMMAND psx_fmc_native_service_tests)
  foreach(component IN ITEMS profile protocol service)
    add_executable(psx_fmc_${component}_tests tests/fmc_${component}_tests.cpp)
    target_link_libraries(psx_fmc_${component}_tests PRIVATE psx_situation)
    add_test(NAME psx_fmc_${component} COMMAND psx_fmc_${component}_tests)
  endforeach()
  # The Java/C++ interchange phase supplies two leased fixture paths. It is
  # intentionally not a no-argument CTest: it never contacts a PSX service.
  add_executable(psx_fmc_interop_tests tests/fmc_interop_tests.cpp)
  target_link_libraries(psx_fmc_interop_tests PRIVATE psx_situation)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../windows-version/VersionInfo.cmake")
psx_stamp_directory("${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_VERSION}")
