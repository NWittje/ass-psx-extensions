cmake_minimum_required(VERSION 3.24)
project(psx_cdu VERSION 0.3.0 LANGUAGES CXX)
include(../windows-version/VersionInfo.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(AssGraphics 1.4.5 EXACT CONFIG REQUIRED)
add_subdirectory(../network-io network-io)
add_library(psx_cdu STATIC src/model.cpp)
target_include_directories(psx_cdu PUBLIC include)
target_compile_definitions(psx_cdu PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
add_library(psx_cdu_codec STATIC src/snapshot.cpp)
target_link_libraries(psx_cdu_codec PUBLIC psx_cdu)
add_library(psx_cdu_blackout STATIC src/blackout.cpp)
target_link_libraries(psx_cdu_blackout PUBLIC psx_cdu_codec)
if(NOT TARGET psx_norm)
 add_subdirectory(../norm norm)
endif()
add_library(psx_cdu_transport STATIC src/transport.cpp src/blackout_pipe.cpp)
target_link_libraries(psx_cdu_transport PUBLIC psx_cdu_codec psx_cdu_blackout psx_norm psx_network_io advapi32)
add_library(psx_cdu_glyphs STATIC src/glyphs.cpp)
target_link_libraries(psx_cdu_glyphs PUBLIC psx_cdu ass::graphics)
add_library(psx_cdu_render STATIC src/render.cpp src/surface.cpp)
target_link_libraries(psx_cdu_render PUBLIC psx_cdu_glyphs)
add_executable(psx-cdu-qualify src/qualify.cpp src/verify.cpp src/surface_verify.cpp)
add_executable(psx-cdu-transport src/transport_main.cpp)
target_link_libraries(psx-cdu-transport PRIVATE psx_cdu_transport psx_cdu_render psapi)
add_executable(cdu_transport_test tests/transport_test.cpp)
target_link_libraries(cdu_transport_test PRIVATE psx_cdu_transport psx_cdu_render psapi)
add_executable(cdu_main_source_test tests/main_source_test.cpp)
target_link_libraries(cdu_main_source_test PRIVATE psx_cdu_transport)
add_test(NAME cdu_main_source_refresh COMMAND cdu_main_source_test)
set_tests_properties(cdu_main_source_refresh PROPERTIES TIMEOUT 15 RUN_SERIAL TRUE)
add_executable(cdu_blackout_test tests/blackout_test.cpp)
target_link_libraries(cdu_blackout_test PRIVATE psx_cdu_transport)
target_link_libraries(psx-cdu-qualify PRIVATE psx_cdu_render psx_cdu_transport windowscodecs ole32)
target_compile_options(psx-cdu-qualify PRIVATE /W4 /utf-8)
foreach(target psx_cdu psx_cdu_codec psx_cdu_blackout psx_norm psx_cdu_transport psx_cdu_glyphs psx_cdu_render psx-cdu-qualify psx-cdu-transport cdu_transport_test cdu_blackout_test)
  target_compile_options(${target} PRIVATE /Zi /W4 /utf-8)
endforeach()
target_link_options(psx-cdu-qualify PRIVATE /DEBUG)
foreach(target psx-cdu-transport cdu_transport_test cdu_blackout_test)
  target_link_options(${target} PRIVATE /DEBUG)
  add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:ass::graphics>" "$<TARGET_FILE_DIR:${target}>" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../norm/vendor/norm/runtime" "$<TARGET_FILE_DIR:${target}>")
endforeach()
add_custom_command(TARGET psx-cdu-qualify POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:ass::graphics>" "$<TARGET_FILE_DIR:psx-cdu-qualify>")
enable_testing()
add_test(NAME cdu_blackout COMMAND cdu_blackout_test)
set_tests_properties(cdu_blackout PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
add_test(NAME cdu_contract COMMAND psx-cdu-qualify verify "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/contract.fixture")
add_test(NAME cdu_captured_contract COMMAND psx-cdu-qualify verify "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/captured-cpt.fixture")
add_test(NAME cdu_render COMMAND psx-cdu-qualify render "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/pages.psx" "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>" --expect-all-colours)
set_tests_properties(cdu_contract cdu_render PROPERTIES TIMEOUT 30)
find_program(PWSH pwsh REQUIRED)
add_test(NAME cdu_observer COMMAND "${PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/observer.ps1" -Executable "$<TARGET_FILE:psx-cdu-qualify>" -Output "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/observer")
set_tests_properties(cdu_observer PROPERTIES TIMEOUT 30)
add_test(NAME cdu_codec_render COMMAND cdu_transport_test codec "${CMAKE_CURRENT_SOURCE_DIR}/fixtures" "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/transport-render")
add_test(NAME cdu_transport_lifecycle COMMAND cdu_transport_test lifecycle)
add_test(NAME cdu_norm_pressure COMMAND cdu_transport_test pressure)
add_test(NAME cdu_norm_loopback COMMAND cdu_transport_test network 127.0.0.1 127.0.0.1 1 0 "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/norm-loopback.json")
foreach(pair "multicast;0" "loss1;1" "loss10;10")
  list(GET pair 0 suffix)
  list(GET pair 1 loss)
  add_test(NAME cdu_norm_${suffix} COMMAND "${PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/norm-network.ps1" -Executable "$<TARGET_FILE:cdu_transport_test>" -Report "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/norm-${suffix}.json" -Loss ${loss})
endforeach()
set_tests_properties(cdu_norm_loopback cdu_norm_multicast cdu_norm_loss1 cdu_norm_loss10 PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
add_test(NAME cdu_norm_processes COMMAND "${PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/norm-processes.ps1" -Executable "$<TARGET_FILE:psx-cdu-transport>" -Fixtures "${CMAKE_CURRENT_SOURCE_DIR}/fixtures" -Output "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/norm-processes")
set_tests_properties(cdu_norm_processes PROPERTIES TIMEOUT 60 RUN_SERIAL TRUE)
add_test(NAME cdu_norm_startup COMMAND "${PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/norm-startup.ps1" -Executable "$<TARGET_FILE:psx-cdu-transport>")
add_test(NAME cdu_norm_ordering COMMAND "${PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/norm-network.ps1" -Executable "$<TARGET_FILE:cdu_transport_test>" -Report "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/norm-ordering.json" -Operation ordering)
set_tests_properties(cdu_norm_ordering PROPERTIES TIMEOUT 20 RUN_SERIAL TRUE)
add_test(NAME cdu_norm_psx_mock COMMAND "${PWSH}" -NoProfile -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/norm-psx-mock.ps1" -Executable "$<TARGET_FILE:psx-cdu-transport>" -Output "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/norm-psx-mock")
set_tests_properties(cdu_norm_psx_mock PROPERTIES TIMEOUT 30 RUN_SERIAL TRUE)

add_test(NAME cdu_surface_render COMMAND psx-cdu-qualify surface-render "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/pages.psx" "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/surface")

# Own these tools even when composed into a host or client build.
psx_stamp_directory("${CMAKE_CURRENT_SOURCE_DIR}" "${psx_cdu_VERSION}")
