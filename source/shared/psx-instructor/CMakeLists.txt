cmake_minimum_required(VERSION 3.30)
project(psx_instructor LANGUAGES CXX)
if(NOT TARGET psx_network_io)
 add_subdirectory(../network-io network-io)
endif()
add_library(psx_instructor STATIC src/communicator.cpp src/reposition.cpp src/airport_database.cpp src/air_data.cpp src/navigation_selection.cpp src/performance_selection.cpp src/weather.cpp src/environment.cpp src/civil_time.cpp src/settings_exchange.cpp src/failures.cpp src/tcas.cpp src/runway_selection.cpp)
target_sources(psx_instructor PRIVATE src/weather_events.cpp src/crash_reset.cpp src/instructor_metadata.cpp)
target_sources(psx_instructor PRIVATE src/weight_service.cpp)
target_sources(psx_instructor PRIVATE src/status_display.cpp)
target_sources(psx_instructor PRIVATE src/maximum_speed.cpp)
target_include_directories(psx_instructor PUBLIC include)
target_compile_features(psx_instructor PUBLIC cxx_std_20)
target_compile_options(psx_instructor PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(psx_instructor PUBLIC psx_network_io iphlpapi)
