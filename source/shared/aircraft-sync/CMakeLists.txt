cmake_minimum_required(VERSION 3.24)
if(NOT TARGET psx_motion)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../motion" "${CMAKE_CURRENT_BINARY_DIR}/motion")
endif()
add_library(psx_aircraft_sync STATIC src/protocol.cpp)
target_include_directories(psx_aircraft_sync PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_link_libraries(psx_aircraft_sync PUBLIC psx_motion)
target_compile_features(psx_aircraft_sync PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(psx_aircraft_sync PRIVATE /W4 /WX /permissive- /utf-8)
endif()
