cmake_minimum_required(VERSION 3.24)
if(NOT TARGET psx_motion)
 add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../motion" "${CMAKE_CURRENT_BINARY_DIR}/motion")
endif()
add_library(psx_environment STATIC src/protocol.cpp src/presentation.cpp)
target_include_directories(psx_environment PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_link_libraries(psx_environment PUBLIC psx_motion)
target_compile_features(psx_environment PUBLIC cxx_std_20)
set_property(TARGET psx_environment PROPERTY CXX_SCAN_FOR_MODULES OFF)
if(WIN32)
 if(NOT TARGET psx_non_normal)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../non-normal" "${CMAKE_CURRENT_BINARY_DIR}/non-normal")
 endif()
 target_sources(psx_environment PRIVATE src/visibility_client.cpp)
 target_link_libraries(psx_environment PUBLIC psx_non_normal)
endif()
if(MSVC)
 target_compile_options(psx_environment PRIVATE /W4 /WX /permissive- /utf-8)
endif()
