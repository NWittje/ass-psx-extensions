cmake_minimum_required(VERSION 3.30)
project(psx_non_normal LANGUAGES CXX)
add_library(psx_non_normal STATIC src/protocol.cpp src/pipe.cpp src/client.cpp)
target_include_directories(psx_non_normal PUBLIC include "${CMAKE_CURRENT_SOURCE_DIR}/../windows-runtime/include")
target_compile_features(psx_non_normal PUBLIC cxx_std_20)
set_property(TARGET psx_non_normal PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(psx_non_normal PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(psx_non_normal PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(psx_non_normal PUBLIC advapi32 bcrypt)
if(BUILD_TESTING OR PSX_NON_NORMAL_TESTS)
 enable_testing()
 add_executable(psx_non_normal_tests tests/protocol.cpp)
 target_link_libraries(psx_non_normal_tests PRIVATE psx_non_normal)
 add_test(NAME psx_non_normal_protocol COMMAND psx_non_normal_tests)
endif()
