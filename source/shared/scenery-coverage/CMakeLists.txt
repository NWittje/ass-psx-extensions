cmake_minimum_required(VERSION 3.24)
if(TARGET psx_scenery_coverage)
 return()
endif()
if(NOT TARGET psx_non_normal)
 add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../non-normal" "${CMAKE_CURRENT_BINARY_DIR}/coverage-non-normal")
endif()
add_library(psx_scenery_coverage STATIC src/protocol.cpp src/client.cpp)
target_include_directories(psx_scenery_coverage PUBLIC include)
target_compile_features(psx_scenery_coverage PUBLIC cxx_std_20)
target_compile_definitions(psx_scenery_coverage PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(psx_scenery_coverage PRIVATE /W4 /WX /utf-8 /EHsc)
target_link_libraries(psx_scenery_coverage PUBLIC psx_non_normal)
