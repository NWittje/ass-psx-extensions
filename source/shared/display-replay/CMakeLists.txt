cmake_minimum_required(VERSION 3.30)
project(psx_display_replay VERSION 0.1.0 LANGUAGES CXX)
add_library(psx_display_replay STATIC src/model.cpp)
target_compile_features(psx_display_replay PUBLIC cxx_std_20)
target_include_directories(psx_display_replay PUBLIC include ../../modules/psx-exporter/include)
target_link_libraries(psx_display_replay PUBLIC ass::graphics)
enable_testing()

target_link_libraries(psx-instrument-replay-test PRIVATE psx_display_replay)
target_compile_definitions(psx-instrument-replay-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME instrument_replay_contract COMMAND psx-instrument-replay-test)
