cmake_minimum_required(VERSION 3.30)
if(TARGET psx_display_transport)
 return()
endif()
if(NOT TARGET psx_norm)
 add_subdirectory(../norm norm)
endif()
if(NOT TARGET psx_network_io)
 add_subdirectory(../network-io network-io)
endif()
add_library(psx_display_transport STATIC src/wire.cpp src/local.cpp src/publisher.cpp src/clock.cpp src/diagnostics.cpp src/tcp_wire.cpp src/tcp_server.cpp src/tcp_client.cpp src/tcp_publisher.cpp src/timing.cpp src/instrument.cpp)
target_include_directories(psx_display_transport PUBLIC include ../display-replay/include ../../modules/psx-exporter/include)
target_compile_features(psx_display_transport PUBLIC cxx_std_20)
target_compile_definitions(psx_display_transport PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(psx_display_transport PRIVATE /W4 /Zi /utf-8 /EHsc)
target_link_libraries(psx_display_transport PUBLIC psx_norm psx_network_io advapi32 bcrypt ws2_32)
add_executable(psx-du-network-qualify tests/qualify.cpp)
target_link_libraries(psx-du-network-qualify PRIVATE psx_display_transport psapi)
target_compile_options(psx-du-network-qualify PRIVATE /W4 /Zi /utf-8 /EHsc)
target_link_options(psx-du-network-qualify PRIVATE /DEBUG /INCREMENTAL:NO)
add_custom_command(TARGET psx-du-network-qualify POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../norm/vendor/norm/runtime" "$<TARGET_FILE_DIR:psx-du-network-qualify>")
enable_testing()
foreach(reader local timing)
 add_executable(psx-du-${reader}-io-fault tests/pipe_io_fault.cpp)
 target_link_libraries(psx-du-${reader}-io-fault PRIVATE psx_display_transport)
 target_compile_options(psx-du-${reader}-io-fault PRIVATE /W4 /Zi /utf-8 /EHsc)
 if(reader STREQUAL "local")
  target_compile_definitions(psx-du-${reader}-io-fault PRIVATE PSX_TEST_LOCAL_IO)
 endif()
 add_test(NAME du_${reader}_io_fault COMMAND psx-du-${reader}-io-fault)
 set_tests_properties(du_${reader}_io_fault PROPERTIES TIMEOUT 5)
endforeach()
add_executable(psx-du-tcp-qualify tests/tcp_qualify.cpp)
add_executable(psx-du-instrument-qualify tests/instrument.cpp)
target_link_libraries(psx-du-instrument-qualify PRIVATE psx_display_transport)
add_test(NAME du_instrument_transport COMMAND psx-du-instrument-qualify "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/instrument-transport")
set_tests_properties(du_instrument_transport PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
add_executable(psx-du-timing-collect tests/timing_collect.cpp)
target_link_libraries(psx-du-timing-collect PRIVATE psx_display_transport)
target_link_libraries(psx-du-tcp-qualify PRIVATE psx_display_transport)
target_compile_options(psx-du-tcp-qualify PRIVATE /W4 /Zi /utf-8 /EHsc)
target_link_options(psx-du-tcp-qualify PRIVATE /DEBUG /INCREMENTAL:NO)
foreach(mode wire payloads cycles faults wildcard)
 add_test(NAME du_tcp_${mode} COMMAND psx-du-tcp-qualify ${mode} "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/tcp-${mode}.jsonl")
 set_tests_properties(du_tcp_${mode} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
endforeach()
add_executable(psx-du-lifecycle-qualify tests/lifecycle.cpp)
add_executable(psx-du-native-reproducer tests/native_reproducer.cpp)
target_include_directories(psx-du-native-reproducer PRIVATE ../norm/vendor/norm/include)
target_compile_features(psx-du-native-reproducer PRIVATE cxx_std_20)
target_compile_definitions(psx-du-native-reproducer PRIVATE NORM_USE_DLL WIN32 NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(psx-du-native-reproducer PRIVATE /W4 /Zi /utf-8 /EHsc)
target_link_options(psx-du-native-reproducer PRIVATE /DEBUG /INCREMENTAL:NO)
target_link_libraries(psx-du-native-reproducer PRIVATE ws2_32)
add_custom_command(TARGET psx-du-native-reproducer POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../norm/vendor/norm/runtime" "$<TARGET_FILE_DIR:psx-du-native-reproducer>")
target_link_libraries(psx-du-lifecycle-qualify PRIVATE psx_display_transport)
target_compile_options(psx-du-lifecycle-qualify PRIVATE /W4 /Zi /utf-8 /EHsc)
target_link_options(psx-du-lifecycle-qualify PRIVATE /DEBUG /INCREMENTAL:NO)
add_custom_command(TARGET psx-du-lifecycle-qualify POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../norm/vendor/norm/runtime" "$<TARGET_FILE_DIR:psx-du-lifecycle-qualify>")
option(PSX_DU_NORM_QUALIFICATION "Explicit comparison of historical DU NORM lifecycle including known native fault" OFF)
if(PSX_DU_NORM_QUALIFICATION)
foreach(mode queues cycles payloads)
 add_test(NAME du_lifecycle_${mode} COMMAND psx-du-lifecycle-qualify ${mode} "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/lifecycle-${mode}")
 set_tests_properties(du_lifecycle_${mode} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
endforeach()
endif()
foreach(mode wire network clock)
 if(mode STREQUAL "network" AND NOT PSX_DU_NORM_QUALIFICATION)
  continue()
 endif()
 add_test(NAME du_transport_${mode} COMMAND psx-du-network-qualify ${mode} "${CMAKE_BINARY_DIR}/evidence/$<CONFIG>/transport-${mode}.json")
 set_tests_properties(du_transport_${mode} PROPERTIES TIMEOUT 40 RUN_SERIAL TRUE)
endforeach()
