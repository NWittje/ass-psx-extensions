cmake_minimum_required(VERSION 3.24)
if(NOT TARGET psx_traffic_source)
 add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../traffic-source" "${CMAKE_CURRENT_BINARY_DIR}/traffic-source")
endif()
add_library(psx_traffic_visual STATIC src/protocol.cpp src/presentation.cpp)
target_include_directories(psx_traffic_visual PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include")
target_link_libraries(psx_traffic_visual PUBLIC psx_traffic_source)
target_compile_features(psx_traffic_visual PUBLIC cxx_std_17)
if(MSVC)
 target_compile_options(psx_traffic_visual PRIVATE /W4 /WX /permissive- /utf-8)
endif()
