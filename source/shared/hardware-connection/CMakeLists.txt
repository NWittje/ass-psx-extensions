cmake_minimum_required(VERSION 3.24)
if(NOT TARGET psx_hardware_link)
  add_subdirectory(../hardware-link hardware-link)
endif()
add_library(psx_hardware_host STATIC src/host_session.cpp)
target_include_directories(psx_hardware_host PUBLIC include)
target_link_libraries(psx_hardware_host PUBLIC psx_hardware_link)
target_compile_features(psx_hardware_host PUBLIC cxx_std_20)
if(WIN32)
  add_library(psx_hardware_connection STATIC src/connection.cpp src/serial_transport.cpp)
  target_link_libraries(psx_hardware_connection PUBLIC psx_hardware_host PRIVATE setupapi cfgmgr32 advapi32 bcrypt uuid)
  target_compile_definitions(psx_hardware_connection PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  # Compile only the existing 12-byte profile codec, without the firmware project's artwork/test graph.
  add_library(psx_trim_connection STATIC src/trim_connection.cpp ../../modules/psx-trim-display/core/protocol.cpp)
  target_include_directories(psx_trim_connection PUBLIC ../../modules/psx-trim-display/include)
  target_link_libraries(psx_trim_connection PUBLIC psx_hardware_connection)
endif()
